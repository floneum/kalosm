FRESH CHECKOUT: CAPTURE A LOCAL NUMERICAL REFERENCE

The PR publishes comparison summaries, not raw reference/candidate vectors.
Build the profiling example before this step. On another machine, set the
PROFILE_MMAP_WEIGHTS path in copies of both configuration JSON files to your
original Q4_K_M GGUF, and use those copies for all runs. This bounded capture
uses the conservative baseline on the same binary and runs sequentially.

export MODEL=/absolute/path/Meta-Llama-3.1-8B-Instruct-Q4_K_M.gguf
python3 - <<'CAPTURE'
import json, os, sys
from pathlib import Path
sys.path.insert(0, "fusor/fusor-visualize/tools")
from benchmark_guard import run_guarded
out = Path("/tmp/llama-reference-new")
out.mkdir(exist_ok=False)
model = Path(os.environ["MODEL"]).resolve()
env = {k: v for k, v in os.environ.items()
       if not k.startswith(("FUSOR_", "PROFILE_", "LAB_"))}
env.update(json.loads(Path("reports/llama-decode-2x/webgpu-memory-bounded/segmented-host/baseline-scoped.json").read_text()))
env.update(PROFILE_MMAP_WEIGHTS=str(model), PROFILE_PROMPT="long",
           PROFILE_PROMPT_REPEATS="8", FUSOR_EXPLORE_EPS="0",
           PROFILE_LOGITS_DIR=str(out / "reference-logits"))
with (out / "reference.jsonl").open("w") as stdout, (out / "ab-reference.log").open("w") as stderr:
    run_guarded([str(Path("target/debug/examples/profile_kernel_share").resolve()),
                 str(model), "1", "8"], cwd=Path.cwd(), env=env,
                stdout=stdout, stderr=stderr, model_bytes=model.stat().st_size,
                timeout=600, report_path=out / "resources.jsonl",
                poll_seconds=.1, record_child_rss=True)
CAPTURE
